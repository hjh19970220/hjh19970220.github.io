 void submitRecharge(int cents,int days,String ref){new Thread(()->{try{JSONObject b=new JSONObject();b.put("p_amount_cents",cents);b.put("p_membership_days",days);b.put("p_payment_reference",ref);postJson(API+"/rest/v1/rpc/hao_app_submit_manual_recharge",b,true);runOnUiThread(()->Toast.makeText(this,"充值申请已提交，等待人工核验",Toast.LENGTH_LONG).show());loadAccount();}catch(Exception e){runOnUiThread(()->Toast.makeText(this,"提交失败："+e.getMessage(),Toast.LENGTH_LONG).show());}}).start();}
 void ordersDialog(){StringBuilder s=new StringBuilder();for(int i=0;i<orders.length();i++){JSONObject o=orders.optJSONObject(i);s.append("¥").append(String.format(Locale.CHINA,"%.2f",o.optLong("amount_cents")/100.0)).append("  ").append(statusCn(o.optString("status"))).append("\n").append(o.optString("created_at")).append("\n\n");}if(s.length()==0)s.append("暂无充值记录");new AlertDialog.Builder(this).setTitle("充值记录").setMessage(s.toString()).setPositiveButton("关闭",null).show();}
 void adminOrders(){new Thread(()->{try{JSONArray a=postArray(API+"/rest/v1/rpc/hao_app_admin_pending_recharges",new JSONObject(),true);runOnUiThread(()->{if(a.length()==0){Toast.makeText(this,"暂无待审核订单",Toast.LENGTH_SHORT).show();return;}String[] it=new String[a.length()];for(int i=0;i<a.length();i++){JSONObject o=a.optJSONObject(i);it[i]=o.optString("display_name")+"  ¥"+String.format(Locale.CHINA,"%.2f",o.optLong("amount_cents")/100.0)+"  "+o.optInt("membership_days")+"天";}new AlertDialog.Builder(this).setTitle("待审核充值").setItems(it,(d,which)->adminApproveDialog(a.optJSONObject(which))).show();});}catch(Exception e){runOnUiThread(()->Toast.makeText(this,"读取失败："+e.getMessage(),Toast.LENGTH_LONG).show());}}).start();}
 void adminApproveDialog(JSONObject o){new AlertDialog.Builder(this).setTitle("确认充值").setMessage("用户："+o.optString("display_name")+"\n金额：¥"+String.format(Locale.CHINA,"%.2f",o.optLong("amount_cents")/100.0)+"\n会员："+o.optInt("membership_days")+"天\n付款备注："+o.optString("payment_reference")).setNegativeButton("拒绝",(d,w)->adminAct("hao_app_admin_reject_recharge",o.optString("order_id"))).setPositiveButton("确认到账",(d,w)->adminAct("hao_app_admin_approve_recharge",o.optString("order_id"))).show();}
 void adminAct(String fn,String id){new Thread(()->{try{JSONObject b=new JSONObject();b.put("p_order_id",id);b.put("p_review_note","豪竞App人工审核");postJson(API+"/rest/v1/rpc/"+fn,b,true);runOnUiThread(()->Toast.makeText(this,"处理完成",Toast.LENGTH_SHORT).show());loadAccount();}catch(Exception e){runOnUiThread(()->Toast.makeText(this,"处理失败："+e.getMessage(),Toast.LENGTH_LONG).show());}}).start();}
