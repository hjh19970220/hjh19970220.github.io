 void auth(String em,String pw,boolean reg,String name){new Thread(()->{try{JSONObject b=new JSONObject();b.put("email",em);b.put("password",pw);if(reg&&!name.isEmpty()){JSONObject m=new JSONObject();m.put("display_name",name);b.put("data",m);}JSONObject r=postJson(API+(reg?"/auth/v1/signup":"/auth/v1/token?grant_type=password"),b,false);if(reg&&r.optString("access_token").isEmpty()){runOnUiThread(()->Toast.makeText(this,"注册成功，请按邮箱提示确认后登录",Toast.LENGTH_LONG).show());return;}saveSession(r,em);loadAccount();runOnUiThread(()->{Toast.makeText(this,reg?"注册并登录成功":"登录成功",Toast.LENGTH_SHORT).show();render();});}catch(Exception ex){runOnUiThread(()->Toast.makeText(this,"操作失败："+ex.getMessage(),Toast.LENGTH_LONG).show());}}).start();}

 void saveSession(JSONObject r,String em){token=r.optString("access_token");refresh=r.optString("refresh_token");JSONObject u=r.optJSONObject("user");uid=u==null?"":u.optString("id");email=em;sp.edit().putString("token",token).putString("refresh",refresh).putString("uid",uid).putString("email",email).apply();}
 void logout(){token=refresh=uid=email="";profile=new JSONObject();wallet=new JSONObject();orders=new JSONArray();sp.edit().clear().apply();render();}
